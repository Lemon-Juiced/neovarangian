##################################################
# Neo-Varangian: Grand Varangian Venture values
##################################################

# Same-culture, same-faith core mainland Scandinavian counties in the attacker's realm.
nv_grand_va_eligible_counties = {
	value = 0
	scope:attacker = {
		every_sub_realm_county = {
			limit = {
				title_province = { geographical_region = dlc_fp1_region_core_mainland_scandinavia }
				culture = scope:attacker.culture
				faith = scope:attacker.faith
			}
			add = 1
		}
	}
	# Max eligible counties that contribute to the host (base CB caps at 10).
	max = 20
}

# 1 stack = 100 men. At the cap (20 counties): 9 VV, 9 Bondi, 14 Vigmen, 5 Huscarls, 3000 levies.
# Base CB at its cap (10 counties): 3 VV, 4 Bondi, 8 Vigmen, 4 Huscarls, 100 levies.
nv_grand_va_varangian_veterans_stacks = {
	value = nv_grand_va_eligible_counties
	divide = 4
	floor = yes
	add = 4
}

nv_grand_va_bondi_stacks = {
	value = nv_grand_va_eligible_counties
	divide = 4
	floor = yes
	add = 4
}

nv_grand_va_vigmen_stacks = {
	value = nv_grand_va_eligible_counties
	divide = 2
	floor = yes
	add = 4
}

nv_grand_va_huscarl_stacks = {
	value = nv_grand_va_eligible_counties
	divide = 4
	floor = yes
}

nv_grand_va_levies = {
	value = nv_grand_va_eligible_counties
	multiply = 100
	add = 1000
}

# Prestige cost: same as the base CB's 1000. No repeat-use scaling since it is once per lifetime.
nv_grand_va_cb_prestige_cost = {
	add = { value = 1000 desc = "CB_BASE_COST" }
	scope:attacker = {
		if = {
			limit = { is_ai = yes }
			multiply = 0.25
		}
	}
}
