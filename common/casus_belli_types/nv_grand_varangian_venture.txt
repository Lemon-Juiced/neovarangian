##################################################
# Neo-Varangian: Grand Varangian Venture
#
# A King-only take on the base-game Varangian Adventure (fp1_varangian_adventurer_conquest).
#  - Same culture/era/prestige/targeting requirements as the base CB.
#  - Kings only: no counts, dukes, emperors or hegemons.
#  - Once per lifetime (consumed on declaration, win or lose).
#  - Raises a larger host than the base CB.
#  - On victory the conquered duchy is elevated into a new Kingdom named after it
#    (e.g. Normandy -> Kingdom of Normandy), with the duchy moved de jure under it.
##################################################

nv_grand_varangian_venture = {
	icon = fp1_varangian_adventurer_conquest
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		culture = {
			has_innovation = innovation_varangian_adventurers
			# Can only be fired in the Tribal Era (same as the base CB).
			NOT = { has_cultural_era_or_later = culture_era_early_medieval }
		}
		OR = {
			# Players can always use this.
			is_ai = no
			# The AI has the same stricter requirements as the base CB.
			AND = {
				OR = {
					AND = {
						has_trait = ambitious
						calc_true_if = {
							amount >= 2
							has_trait = greedy
							has_trait = brave
							has_trait = arrogant
							has_trait = adventurer
						}
					}
					has_game_rule = fp1_scandi_adventurers_apocalyptic
				}
				any_realm_county = {
					count = all
					title_province = { geographical_region = world_europe_north }
				}
			}
		}
		NOT = {
			government_has_flag = government_is_landless_adventurer
		}
	}

	allowed_for_character_display_regardless = {
		top_liege = this
		# Kings only.
		custom_tooltip = {
			text = nv_grand_va_kings_only_tt
			highest_held_title_tier = tier_kingdom
		}
		# Once per lifetime.
		custom_tooltip = {
			text = nv_grand_va_once_per_life_tt
			NOT = { has_variable = nv_grand_va_used }
		}
		trigger_if = {
			limit = { is_ai = no }
			trigger_if = {
				limit = {
					culture = {
						has_cultural_parameter = cb_lower_prestige_level_requirement
					}
				}
				prestige_level >= 1
			}
			trigger_else = {
				prestige_level >= 2
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					culture = {
						has_cultural_parameter = cb_lower_prestige_level_requirement
					}
				}
				prestige_level >= 0
			}
			trigger_else = {
				prestige_level >= 1
			}
		}
		is_at_war = no
		tgp_japan_offensive_wars_ban_trigger = yes
	}

	allowed_against_character = {
		# Cannot adventure against other adventuring cultures.
		NOT = {
			culture = { has_innovation = innovation_varangian_adventurers }
		}
		# Players vs. AI filtering.
		OR = {
			is_ai = yes
			any_sub_realm_duchy = {
				count >= 2
				save_temporary_scope_as = player_controlled_duchy
				custom_description = {
					text = SUPPRESS
					scope:defender = { completely_controls = scope:player_controlled_duchy }
				}
			}
		}
		# Scope:defender cannot be bordering your realm.
		scope:defender = {
			NOT = { character_is_land_realm_neighbor = scope:attacker }
		}
	}
	target_titles = all
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	ai_can_target_all_titles = { always = yes }
	ai_score_mult = {
		value = varangian_adventure_ai_targeting_value
		multiply = frankokratia_leader_protection_value
	}

	valid_to_start = {
	}

	should_invalidate = {
		OR = {
			NOT = {
				any_in_list = {
					list = target_titles
					any_in_de_jure_hierarchy = {
						tier = tier_county
						holder = {
							target_is_same_character_or_above = scope:defender
						}
					}
				}
			}
			scope:attacker = {
				OR = {
					top_liege != this
					NOT = { highest_held_title_tier = tier_kingdom }
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message

	on_invalidated = {}

	cost = {
		piety = {
			add = fp1_varangian_adventure_base_piety_cost
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = nv_grand_va_cb_prestige_cost
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
		scope:attacker = {
			# Consume the once-per-lifetime use.
			custom_tooltip = nv_grand_va_consumed_tt
			set_variable = { name = nv_grand_va_used value = yes }

			capital_barony = { save_scope_as = capital_barony }

			if = {
				limit = {
					OR = {
						fp1_va_eligible_for_scandinavian_troops_trigger = yes
						dynasty ?= { has_dynasty_perk = fp1_adventure_legacy_1 }
					}
				}
				send_interface_toast = {
					type = msg_troops_gained
					title = nv_grand_varangian_venture.troops_raised.tt
					left_icon = scope:attacker

					## Royal Host: scales with same-culture, same-faith core Scandinavian counties (cap 20, vs 10 for the base CB).
					if = {
						limit = { fp1_va_eligible_for_scandinavian_troops_trigger = yes }
						spawn_army = {
							men_at_arms = {
								type = varangian_veterans
								stacks = nv_grand_va_varangian_veterans_stacks
							}
							men_at_arms = {
								type = bondi
								stacks = nv_grand_va_bondi_stacks
							}
							men_at_arms = {
								type = vigmen
								stacks = nv_grand_va_vigmen_stacks
							}
							levies = nv_grand_va_levies
							inheritable = yes
							location = scope:attacker.capital_province
							war = scope:war
							war_keep_on_attacker_victory = yes
							save_scope_as = new_army
							name = nv_grand_va_event_troops
						}
						if = {
							limit = { nv_grand_va_huscarl_stacks >= 1 }
							spawn_army = {
								men_at_arms = {
									type = huscarl
									stacks = nv_grand_va_huscarl_stacks
								}
								inheritable = yes
								location = scope:attacker.capital_province
								war = scope:war
								war_keep_on_attacker_victory = yes
								name = nv_grand_va_event_troops
							}
						}
					}
					## Adventure Legacy perk army (double the base CB's).
					if = {
						limit = {
							dynasty ?= { has_dynasty_perk = fp1_adventure_legacy_1 }
						}
						spawn_army = {
							men_at_arms = {
								type = varangian_veterans
								stacks = 8
							}
							men_at_arms = {
								type = vigmen
								stacks = 8
							}
							men_at_arms = {
								type = bondi
								stacks = 2
							}
							levies = 200
							inheritable = yes
							location = scope:attacker.capital_province
							war = scope:war
							war_keep_on_attacker_victory = yes
							save_scope_as = new_army
							name = nv_grand_va_event_troops
						}
					}
				}
			}
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
	}

	on_victory = {
		add_legitimacy_attacker_victory_effect = yes
		save_scope_value_as = {
			name = suppress_legitimacy_loss
			value = yes
		}

		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		scope:attacker = {
			## First up, hand over counties in the target duchy to scope:attacker.
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = yes
			}
			every_in_list = {
				list = target_titles
				custom_tooltip = CONQUEST_CB_TITLE
				every_in_de_jure_hierarchy = {
					continue = {
						OR = {
							NOT = {
								exists = holder
							}
							NOT = {
								holder = {
									target_is_liege_or_above = scope:defender
								}
							}
							AND = {
								scope:attacker.faith = {
									faith_hostility_level_comparison = {
										prev.holder.faith <= scope:defender.faith
									}
								}
								holder = {
									OR = {
										this = scope:defender
										target_is_liege_or_above = scope:defender
									}
								}
							}
						}
						tier > tier_county
					}
					limit = {
						tier = tier_county
						holder ?= {
							OR = {
								this = scope:defender
								target_is_liege_or_above = scope:defender
							}
						}
					}
					if = {
						limit = {
							OR = {
								scope:attacker.faith = {
									faith_hostility_level_comparison = { prev.holder.faith >= scope:defender.faith }
								}
								holder.primary_title.tier >= scope:attacker.primary_title.tier
							}
						}
						add_to_temporary_list = titles_taken
					}
					else = {
						holder = { add_to_temporary_list = vassals_taken }
					}
				}
			}
			every_in_list = {
				list = titles_taken
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
					take_baronies = yes
				}
			}
			every_in_list = {
				list = vassals_taken
				change_liege = {
					liege = scope:attacker
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			## Grab the appropriate duchy.
			random_in_list = {
				list = target_titles
				limit = { tier = tier_duchy }
				save_scope_as = target_duchy
			}
			## Remember the duchy's current de jure empire so the new kingdom can slot in beneath it.
			if = {
				limit = { exists = scope:target_duchy.empire }
				scope:target_duchy.empire = { save_scope_as = nv_old_empire }
			}
			## If the duchy doesn't already exist, create it.
			if = {
				limit = {
					any_in_list = {
						list = target_titles
						tier = tier_duchy
						is_title_created = no
					}
				}
				random_in_list = {
					list = target_titles
					limit = { tier = tier_duchy }
					save_scope_as = new_duchy
				}
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:new_duchy = {
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			## Else if it does & is a part of scope:defender's sub-realm, usurp it.
			else_if = {
				limit = {
					any_in_list = {
						list = target_titles
						tier = tier_duchy
						holder = {
							OR = {
								this = scope:defender
								any_liege_or_above = { this = scope:defender }
							}
						}
					}
				}
				random_in_list = {
					list = target_titles
					limit = { tier = tier_duchy }
					save_scope_as = new_duchy
				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = yes
				}
				scope:new_duchy = {
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			## Else the de jure duchy is held by an outsider: create a titular duchy, as the base CB does.
			else = {
				show_as_tooltip = {
					random_in_list = {
						list = target_titles
						limit = { tier = tier_duchy }
						save_scope_as = new_duchy
					}
					create_title_and_vassal_change = {
						type = created
						save_scope_as = change
						add_claim_on_loss = no
					}
					scope:new_duchy = {
						change_title_holder = {
							holder = scope:attacker
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
				hidden_effect = {
					random_list = {
						25 = {
							create_dynamic_title = {
								tier = duchy
								name = VIKING_STATE_TITLE_NAME
							}
						}
						25 = {
							create_dynamic_title = {
								tier = duchy
								name = NORDIC_STATE_TITLE_NAME
							}
						}
						25 = {
							trigger = {
								any_in_list = {
									list = target_titles
									tier = tier_county
									title_province = {
										geographical_region = world_europe_south_east
										geographical_region = world_europe_east
										geographical_region = world_asia_minor
									}
								}
							}
							create_dynamic_title = {
								tier = duchy
								name = VARYAG_STATE_TITLE_NAME
							}
						}
						25 = {
							create_dynamic_title = {
								tier = duchy
								name = NORSE_STATE_TITLE_NAME
							}
						}
					}
					create_title_and_vassal_change = {
						type = created
						save_scope_as = change
						add_claim_on_loss = no
					}
					scope:new_title = {
						set_destroy_on_gain_same_tier = yes
						set_no_automatic_claims = yes
						set_can_be_named_after_dynasty = no
						change_title_holder = {
							holder = scope:attacker
							change = scope:change
						}
						save_scope_as = new_duchy
					}
					resolve_title_and_vassal_change = scope:change
					scope:new_title = { generate_coa = yes }
				}
			}
			## Set a new capital.
			if = {
				limit = {
					any_in_list = {
						list = target_titles
						this = duchy.title_capital_county
					}
				}
				random_in_list = {
					list = target_titles
					limit = { this = duchy.title_capital_county }
					save_temporary_scope_as = new_capital
					scope:attacker = { set_realm_capital = scope:new_capital }
				}
			}
			else_if = {
				limit = {
					scope:attacker = { government_has_flag = government_is_tribal }
				}
				ordered_in_list = {
					list = target_titles
					limit = { tier = tier_county }
					order_by = total_county_levies
					save_temporary_scope_as = new_capital
					scope:attacker = { set_realm_capital = scope:new_capital }
				}
			}
			else_if = {
				limit = {
					any_in_list = {
						list = target_titles
						tier = tier_county
						uses_county_fertility = yes
					}
				}
				ordered_in_list = {
					list = target_titles
					limit = { tier = tier_county }
					order_by = county_fertility
					save_temporary_scope_as = new_capital
					scope:attacker = { set_realm_capital = scope:new_capital }
				}
			}
			else = {
				ordered_in_list = {
					list = target_titles
					limit = { tier = tier_county }
					order_by = development_level
					save_temporary_scope_as = new_capital
					scope:attacker = { set_realm_capital = scope:new_capital }
				}
			}

			##################################################
			# Neo-Varangian: raise the conquered duchy into a new Kingdom.
			##################################################
			custom_tooltip = nv_grand_varangian_venture.summaries.new_kingdom
			hidden_effect = {
				create_dynamic_title = {
					tier = kingdom
					name = NV_GRAND_VA_KINGDOM_NAME
				}
				scope:new_title = { save_scope_as = nv_new_kingdom }
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:nv_new_kingdom = {
					set_can_be_named_after_dynasty = no
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
				# Look like the duchy it was born from, and sit in the right place de jure.
				scope:nv_new_kingdom = {
					set_coa = scope:new_duchy
					set_color_from_title = scope:new_duchy
					set_capital_county = scope:attacker.capital_county
					if = {
						limit = { exists = scope:nv_old_empire }
						set_de_jure_liege_title = scope:nv_old_empire
					}
				}
				scope:new_duchy = { set_de_jure_liege_title = scope:nv_new_kingdom }
			}

			# Then re-order stuff in the Old Country.
			## Destroy the old kingdom(s) the attacker held (excepting HoF titles and the new kingdom).
			every_held_title = {
				title_tier = kingdom
				limit = {
					NOR = {
						this = scope:nv_new_kingdom
						is_head_of_faith = yes
					}
				}
				save_temporary_scope_as = excess_kingdom
				scope:attacker = { destroy_title = scope:excess_kingdom }
			}
			## Destroy any other ducal titles scope:attacker holds (excepting HoF titles).
			every_held_title = {
				title_tier = duchy
				limit = {
					NOR = {
						this = scope:new_duchy
						is_head_of_faith = yes
					}
				}
				save_temporary_scope_as = excess_duchy
				scope:attacker = { destroy_title = scope:excess_duchy }
			}
			## Sort out vassalised family.
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.family_transfer
			every_vassal_or_below = {
				limit = { dynasty = scope:attacker.dynasty }
				add_to_list = vassalised_family_list
			}
			every_held_title = {
				title_tier = county
				limit = {
					duchy = { is_in_list = target_titles }
				}
				add_to_list = spare_counties_list
			}
			if = {
				limit = {
					any_in_list = {
						list = vassalised_family_list
						count >= 1
					}
				}
				every_in_list = {
					list = vassalised_family_list
					limit = { is_ai = no }
					alternative_limit = { highest_held_title_tier = tier_county }
					save_temporary_scope_as = receiving_vassal
					if = {
						limit = {
							any_in_list = {
								list = spare_counties_list
								this != scope:attacker.capital_county
								holder = scope:attacker
							}
						}
						random_in_list = {
							list = spare_counties_list
							limit = {
								this != scope:attacker.capital_county
								tier = tier_county
								holder = scope:attacker
							}
							save_temporary_scope_as = spare_county
							create_title_and_vassal_change = {
								type = granted
								save_scope_as = change
								add_claim_on_loss = no
							}
							scope:spare_county = {
								change_title_holder = {
									holder = scope:receiving_vassal
									change = scope:change
								}
							}
							resolve_title_and_vassal_change = scope:change
						}
						add_to_list = vassal_family_processed_list
					}
					else = { add_to_list = vassal_family_not_processed_list }
				}
			}
			## Distribute scope:attacker's old domains to peasant vassals of the local culture/faith.
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.old_domains
			every_held_title = {
				title_tier = county
				limit = {
					duchy = {
						this != scope:target_duchy
					}
				}
				save_temporary_scope_as = current_county
				create_character = {
					location = scope:current_county.title_province
					template = old_country_local_warlord_template
					save_temporary_scope_as = local_warlord
				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				hidden_effect = {
					scope:current_county = {
						change_title_holder = {
							holder = scope:local_warlord
							change = scope:change
							take_baronies = yes
						}
					}
				}
				resolve_title_and_vassal_change = scope:change
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = no
				}
				hidden_effect = {
					scope:local_warlord = {
						becomes_independent = { change = scope:change }
						save_scope_as = truce_loc
						add_truce_both_ways = {
							character = scope:attacker
							days = 14600
							name = FP1_VARANGIAN_ADVENTURER_INDEPENDENCE_TRUCE_NAME
						}
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			## Distribute landed family's domains outside of the colony to peasant vassals of the local culture/faith.
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.landed_family_old_domains
			every_in_list = {
				list = vassal_family_processed_list
				every_held_title = {
					title_tier = county
					limit = {
						duchy = {
							this != scope:target_duchy
						}
					}
					save_temporary_scope_as = current_county
					create_character = {
						location = scope:current_county.title_province
						template = old_country_local_warlord_template
						save_temporary_scope_as = local_warlord
					}
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = change
						add_claim_on_loss = no
					}
					hidden_effect = {
						scope:current_county = {
							change_title_holder = {
								holder = scope:local_warlord
								change = scope:change
								take_baronies = yes
							}
						}
					}
					resolve_title_and_vassal_change = scope:change
					create_title_and_vassal_change = {
						type = independency
						save_scope_as = change
						add_claim_on_loss = no
					}
					hidden_effect = {
						scope:local_warlord = {
							becomes_independent = { change = scope:change }
							save_scope_as = truce_loc
							add_truce_both_ways = {
								character = scope:attacker
								days = 14600
								name = FP1_VARANGIAN_ADVENTURER_INDEPENDENCE_TRUCE_NAME
							}
						}
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
			## Give all old vassals outside of the target duchy independence.
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.old_vassals
			every_vassal = {
				limit = {
					NOT = {
						any_held_title = { duchy = scope:target_duchy }
					}
				}
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = no
				}
				hidden_effect = {
					becomes_independent = { change = scope:change }
					save_scope_as = truce_loc
					add_truce_both_ways = {
						character = scope:attacker
						days = 14600
						name = FP1_VARANGIAN_ADVENTURER_INDEPENDENCE_TRUCE_NAME
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			# Sort the governance of the colony.
			if = {
				limit = {
					capital_province = { has_holding_type = tribal_holding }
					NOT = { government_has_flag = government_is_tribal }
				}
				change_government = tribal_government
			}
			else_if = {
				limit = {
					NOT = {
						capital_province = { has_holding_type = tribal_holding }
					}
				}
				change_government = feudal_government
			}
			## Seizing control so things *vaguely* work.
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.control_rebate
			scope:attacker = { trigger_event = fp1_varangian_adventures.0001 }
			## A bigger bounty than the base CB, befitting a king.
			scope:attacker = {
				add_gold = {
					value = varangian_adventures_bounty_amount_value
					multiply = 2
				}
			}
			## Move any remaining newly-landless family to scope:attacker's court.
			every_in_list = {
				list = vassal_family_not_processed_list
				limit = { is_ruler = no }
				save_temporary_scope_as = landless_family
				scope:attacker = { add_courtier = scope:landless_family }
			}
			## House handling, as the base CB.
			if = {
				limit = {
					this != house.house_head
				}
				found_cadet_house_decision_effect = {
					CHARACTER = scope:attacker
					PRESTIGE = 0
				}
			}
			else = {
				house = {
					ordered_house_member = {
						limit = {
							is_ruler = yes
							NOR = {
								this = scope:attacker
								any_liege_or_above = { this = scope:attacker }
							}
						}
						order_by = age
						max = 100
						check_range_bounds = no
						save_scope_as = attacker_house_head
						found_cadet_house_decision_effect = {
							CHARACTER = scope:attacker_house_head
							PRESTIGE = 0
						}
					}
				}
			}
			if = {
				limit = {
					NOR = {
						has_trait = adventurer
						government_has_flag = government_is_nomadic
					}
				}
				add_trait = adventurer
			}
			hidden_effect = {
				every_courtier = { set_location_to_default = yes }
				every_prisoner = { set_location_to_default = yes }
			}
		}

		# Prestige Progress for the Attacker (a grand venture earns major renown).
		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = major_prestige_value
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		if = {
			limit = {
				scope:attacker = {
					any_ally = {
						NOT = { in_diplomatic_range = scope:attacker }
					}
				}
			}
			scope:attacker = {
				every_ally = {
					limit = {
						NOT = { in_diplomatic_range = scope:attacker }
					}
					break_alliance = scope:attacker
				}
			}
		}

		if = {
			limit = {
				exists = scope:attacker.confederation
			}
			scope:attacker.confederation = {
				remove_confederation_member = scope:attacker
			}
		}

		add_truce_attacker_victory_effect = yes

		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes

		mandala_war_victory_effects = yes
	}
