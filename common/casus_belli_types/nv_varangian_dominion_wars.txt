##################################################
# Neo-Varangian: Varangian Dominion wars
#
# nv_varangian_dominion_cb    - Nordokratia / Normanokratia: a Norse or Norman ruler
#                                seizes a Byzantine kingdom and founds the Varangian Empire.
# nv_seize_imperial_duchy_cb  - Seize Imperial Duchy with the Varangian Empire in the Latin
#                                Empire's place, usable by both sides (the base CB only knows
#                                the Latin Empire).
##################################################

nv_varangian_dominion_cb = {
	icon = fp1_varangian_adventurer_conquest
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		# Player-only: only one Varangian Empire can exist, so the AI must not claim it first.
		is_ai = no
		# Unlocked by the "Plan the Taking of Miklagard" decision.
		has_variable = nv_miklagard_ambition
		nv_varangian_culture_trigger = yes
		NOT = { exists = title:e_varangian_empire.holder }
	}

	allowed_for_character_display_regardless = {
		top_liege = this
		highest_held_title_tier >= tier_kingdom
		is_at_war = no
	}

	allowed_against_character = {
		exists = title:e_byzantium.holder
		scope:defender = title:e_byzantium.holder
	}
	target_titles = all
	target_title_tier = kingdom
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	ai_can_target_all_titles = { always = yes }
	ai_score_mult = {
		value = 50
		# The AI goes for Constantinople's kingdom first.
		if = {
			limit = {
				scope:target ?= { any_de_jure_county = { this = title:c_byzantion } }
			}
			multiply = 4
		}
		multiply = frankokratia_leader_protection_value
	}

	valid_to_start = {
		# Must be a Byzantine de jure kingdom where the Emperor's realm holds land.
		scope:target = {
			empire ?= title:e_byzantium
			any_de_jure_county = {
				holder ?= {
					target_is_same_character_or_above = scope:defender
				}
			}
		}
	}

	should_invalidate = {
		OR = {
			NOT = {
				any_in_list = {
					list = target_titles
					any_in_de_jure_hierarchy = {
						tier = tier_county
						holder = {
							target_is_same_character_or_above = scope:defender
						}
					}
				}
			}
			scope:attacker = { top_liege != this }
			exists = title:e_varangian_empire.holder
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message

	on_invalidated = {}

	cost = {
		piety = {
			add = common_cb_impious_piety_cost
			multiply = common_cb_piety_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_victory_effect = yes
		save_scope_value_as = {
			name = suppress_legitimacy_loss
			value = yes
		}

		scope:attacker = { accolade_attacker_war_end_glory_gain_high_effect = yes }

		random_in_list = {
			list = target_titles
			limit = { tier = tier_kingdom }
			save_scope_as = nv_target_kingdom
		}

		# Take the defender's land in the kingdom, as a normal conquest.
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		scope:defender = { ep3_war_loss_adventurer_tt_effect = yes }

		every_in_list = {
			list = target_titles
			custom_tooltip = CONQUEST_CB_TITLE
			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = no
			}
		}

		hidden_effect = {
			every_in_list = {
				list = vassals_taken
				change_liege = {
					liege = scope:attacker
					change = scope:change
				}
			}
		}

		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}

		resolve_title_and_vassal_change = scope:change

		# Found the Varangian Empire and leave the homeland behind.
		custom_tooltip = nv_varangian_dominion.summaries.empire
		scope:attacker = { nv_found_varangian_empire_effect = yes }

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		# Break alliances now out of diplomatic range.
		scope:attacker = {
			every_ally = {
				limit = {
					NOT = { in_diplomatic_range = scope:attacker }
				}
				break_alliance = scope:attacker
			}
		}

		add_truce_attacker_victory_effect = yes

		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }

		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_white_peace_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = county_conquest_cb_defeat_desc_attacker
			}
			desc = county_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_defeat_effect = yes

		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = major_prestige_value
			}
		}

		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
			# The dream is broken: the decision must be taken again to try once more.
			custom_tooltip = nv_miklagard_ambition_lost_tt
			remove_variable = nv_miklagard_ambition
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_defeat_effect = yes

		scope:attacker = {
			save_temporary_scope_as = loser
		}

		on_lost_aggression_war_discontent_loss = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "NV_VARANGIAN_DOMINION_WAR_NAME"
	war_name_base = "NV_VARANGIAN_DOMINION_WAR_NAME_BASE"
	cb_name = "NV_VARANGIAN_DOMINION_CB_NAME"

	interface_priority = 90

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

##################################################
# Seize Imperial Duchy (Varangian)
# Copy of greek_anarchy_cb (Seize Imperial Duchy) with the Varangian Empire in the
# Latin Empire's place. Usable by both sides:
#  - The Varangian Emperor and vassals: any Byzantine/Varangian de jure duchy outside their realm.
#  - The Byzantine Emperor, vassals and Byzantine loyalists: Varangian realm land only.
#  - Anyone else in the old empire: only while Byzantium has no Emperor (as the base CB).
# Every war must involve the Varangian Empire (attacker realm, defender realm or target's
# de jure empire), so it never duplicates the base CB.
##################################################

nv_seize_imperial_duchy_cb = {
	icon = duchy_expansion
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		# The Varangian Empire must exist, or still have de jure land.
		OR = {
			exists = title:e_varangian_empire.holder
			title:e_varangian_empire = {
				any_in_de_jure_hierarchy = { tier <= tier_kingdom }
			}
		}
		# Who may use it.
		OR = {
			# The Varangian side.
			AND = {
				exists = title:e_varangian_empire.holder
				top_liege = title:e_varangian_empire.holder
			}
			# The Byzantine Emperor and his vassals.
			AND = {
				exists = title:e_byzantium.holder
				top_liege = title:e_byzantium.holder
			}
			# Byzantine loyalists anywhere.
			nv_byzantine_loyalist_trigger = yes
			# Anyone at all while Byzantium has no Emperor (as the base CB).
			NOT = { exists = title:e_byzantium.holder }
		}
		# Must have interests within the old empire.
		any_held_title = {
			tier <= tier_empire
			OR = {
				this = title:e_byzantium
				this = title:e_varangian_empire
				empire ?= title:e_byzantium
				empire ?= title:e_varangian_empire
			}
		}
		# Byzantium must still have de jure land.
		title:e_byzantium = {
			any_in_de_jure_hierarchy = { tier <= tier_kingdom }
		}
	}

	allowed_against_character = {
		scope:attacker = {
			NOR = {
				top_liege = scope:defender.top_liege
				liege = scope:defender
			}
		}
		# Must hold land within the old empire.
		scope:defender = {
			any_realm_county = {
				OR = {
					empire ?= title:e_byzantium
					empire ?= title:e_varangian_empire
				}
			}
		}
		# While Byzantium stands, only the Varangians may use this against non-Varangians.
		trigger_if = {
			limit = {
				exists = title:e_byzantium.holder
				NOT = {
					AND = {
						exists = title:e_varangian_empire.holder
						scope:attacker.top_liege = title:e_varangian_empire.holder
					}
				}
			}
			exists = title:e_varangian_empire.holder
			scope:defender.top_liege = title:e_varangian_empire.holder
		}
	}
	target_titles = neighbor_land_or_water
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	ai_score_mult = {
		value = 1
		if = {
			limit = {
				exists = title:e_varangian_empire.holder
				scope:attacker.top_liege = title:e_varangian_empire.holder
			}
			multiply = 5
		}
		else_if = {
			limit = {
				scope:attacker = {
					OR = {
						AND = {
							exists = title:e_byzantium.holder
							top_liege = title:e_byzantium.holder
						}
						culture = culture:greek
						faith = faith:orthodox
					}
				}
			}
			multiply = 3
		}
	}

	valid_to_start = {
		# Must be a duchy of the old Byzantine or the Varangian Empire.
		scope:target = {
			OR = {
				empire ?= title:e_byzantium
				empire ?= title:e_varangian_empire
			}
		}
		# The Varangian Empire must be involved, so this never duplicates the base Seize Imperial Duchy.
		OR = {
			scope:target = { empire ?= title:e_varangian_empire }
			AND = {
				exists = title:e_varangian_empire.holder
				OR = {
					scope:attacker.top_liege = title:e_varangian_empire.holder
					scope:defender.top_liege = title:e_varangian_empire.holder
				}
			}
		}
		# Only players and the extra-motivated should be attacking over water.
		scope:target = {
			any_in_de_jure_hierarchy = {
				tier = tier_county
				OR = {
					scope:attacker = { is_ai = no }
					any_neighboring_county = {
						holder = {
							target_is_same_character_or_above = scope:attacker
						}
					}
					# You serve the 'rightful' emperor.
					AND = {
						exists = title:e_byzantium.holder
						scope:attacker.top_liege = title:e_byzantium.holder
					}
					AND = {
						exists = title:e_varangian_empire.holder
						scope:attacker.top_liege = title:e_varangian_empire.holder
					}
					# Varangians are sea-raiders at heart.
					scope:attacker = { nv_varangian_culture_trigger = yes }
					scope:attacker = {
						government_has_flag = government_is_administrative
						vassal_contract_has_flag = admin_theme_naval
					}
					scope:attacker = {
						ai_greed >= high_positive_ai_value
						ai_boldness >= high_positive_ai_value
					}
				}
			}
		}
	}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = {
						target_is_same_character_or_above = scope:defender
					}
				}
			}
		}
	}

	on_invalidated_desc = msg_religious_war_invalidation_region_message

	on_invalidated = {}

	cost = {
		piety = {
			add = common_cb_impious_piety_cost
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = 0
			if = {
				limit = {
					scope:defender.faith != scope:attacker.faith
				}
				add = {
					value = 150
					desc = CB_COST_DIFFERENT_FAITH
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_victory_effect = yes

		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		# Gain a little army, as Seize Imperial Duchy does.
		hidden_effect = {
			scope:attacker = {
				save_scope_as = army_receiver
				greek_anarchy_army_spawn_effect = yes
			}
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		scope:defender = { ep3_war_loss_adventurer_tt_effect = yes }

		every_in_list = {
			list = target_titles
			custom_tooltip = CONQUEST_CB_TITLE
			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = no
			}
		}

		hidden_effect = {
			every_in_list = {
				list = vassals_taken
				change_liege = {
					liege = scope:attacker
					change = scope:change
				}
			}
		}

		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}

		resolve_title_and_vassal_change = scope:change

		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = medium_prestige_value
			}
		}

		scope:attacker = {
			custom_tooltip = greek_anarchy_troops_tt
			# Varangian conquerors gain dread and plunder, as Latins do with the base CB.
			if = {
				limit = {
					exists = title:e_varangian_empire.holder
					top_liege = title:e_varangian_empire.holder
					nv_byzantine_loyalist_trigger = no
					scope:defender = {
						highest_held_title_tier >= tier_duchy
					}
				}
				add_dread = minor_dread_gain
				add_gold = minor_gold_value
			}
			else = {
				add_prestige = minor_prestige_gain
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes

		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }

		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_white_peace_effect = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = county_conquest_cb_defeat_desc_attacker
			}
			desc = county_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		add_legitimacy_attacker_defeat_effect = yes

		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_defeat_effect = yes

		scope:attacker = {
			save_temporary_scope_as = loser
		}
		on_lost_aggression_war_discontent_loss = yes

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "NV_SEIZE_IMPERIAL_DUCHY_WAR_NAME"
	war_name_base = "NV_SEIZE_IMPERIAL_DUCHY_WAR_NAME_BASE"
	cb_name = "NV_SEIZE_IMPERIAL_DUCHY_CB_NAME"

	interface_priority = 120

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
