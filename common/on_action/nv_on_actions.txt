# Neo-Varangian on_actions.
# Each base on_action below only gets an on_actions list added, so it merges with the base game's version.
yearly_global_pulse = {
	on_actions = { nv_varangian_yearly_pulse }
}

on_birth_child = {
	on_actions = { nv_blood_birth_on_action }
}

on_game_start = {
	on_actions = { nv_blood_game_start_on_action }
}

random_yearly_everyone_pulse = {
	on_actions = { nv_blood_backdate_on_action }
}

# Root is any character, once a year.
# Viking and Varangian are granted from dozens of base-game events, so instead of
# hooking each one, carriers pass the blood traits to existing children here.
nv_blood_backdate_on_action = {
	trigger = {
		OR = {
			has_trait = viking
			has_trait = varangian
			has_trait = nv_blood_of_the_vikings
			has_trait = nv_blood_of_the_varangians
		}
		any_child = { is_alive = yes }
	}
	effect = {
		nv_backdate_blood_traits_effect = yes
	}
}

# Root is the newborn child.
nv_blood_birth_on_action = {
	effect = {
		nv_inherit_blood_traits_effect = yes
	}
}

# No root.
nv_blood_game_start_on_action = {
	effect = {
		nv_seed_blood_traits_effect = yes
	}
}

# Root does not exist (global pulse).
# If the Varangian Empire has fallen, its de jure kingdoms return to Byzantium.
# Mirrors the Latin Empire clean-up in the base game's yearly_global_pulse.
nv_varangian_yearly_pulse = {
	effect = {
		if = {
			limit = {
				title:e_varangian_empire = {
					any_in_de_jure_hierarchy = { tier = tier_kingdom }
				}
				OR = {
					NOT = { exists = title:e_varangian_empire.holder }
					title:e_varangian_empire.holder ?= {
						NOT = {
							any_realm_county = {
								OR = {
									empire ?= title:e_varangian_empire
									empire ?= title:e_byzantium
								}
							}
						}
					}
				}
			}
			title:e_varangian_empire = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_kingdom }
					set_de_jure_liege_title = title:e_byzantium
				}
			}
		}
	}
}
